## Guía de Uso de Helpers

### Ejemplo Completo de Test con Helpers

```typescript
/**
 * Ejemplo completo de test usando todos los helpers
 */
import {
  createTestAdmin,
  loginAsAdmin,
  cleanupTestUsers,
  TEST_ADMIN,
} from '../helpers/auth.helpers';
import {
  cleanTable,
  seedTestClientes,
  countRecords,
} from '../helpers/database.helpers';
import {
  authGet,
  authPost,
  expectPaginatedResponse,
  expectSuccess,
} from '../helpers/request.helpers';

describe('Ejemplo de Test Completo', () => {
  let adminToken: string;

  // Setup: crear usuario admin
  beforeAll(async () => {
    await createTestAdmin();
    adminToken = await loginAsAdmin();
  });

  // Cleanup: limpiar usuario admin
  afterAll(async () => {
    await cleanupTestUsers([TEST_ADMIN.username]);
  });

  // Limpiar tabla antes de cada test
  beforeEach(async () => {
    await cleanTable('clientes');
  });

  describe('GET /api/clientes', () => {
    it('debe retornar lista vacía cuando no hay datos', async () => {
      const response = await authGet('/api/clientes', adminToken);

      expectSuccess(response);
      expectPaginatedResponse(response);
      expect(response.body.data).toHaveLength(0);
    });

    it('debe retornar lista con datos', async () => {
      // Insertar datos de prueba
      await seedTestClientes();

      const response = await authGet('/api/clientes', adminToken);

      expectSuccess(response);
      expect(response.body.data.length).toBe(2);
    });
  });

  describe('POST /api/clientes', () => {
    it('debe crear cliente exitosamente', async () => {
      const clienteData = {
        nombre: 'Cliente Test',
        ruc: '123456789',
        activo: true,
      };

      const response = await authPost('/api/clientes', adminToken, clienteData);

      expectSuccess(response);
      expect(response.body).toHaveProperty('id');
      expect(response.body.nombre).toBe(clienteData.nombre);

      // Verificar que se insertó en BD
      const count = await countRecords('clientes');
      expect(count).toBe(1);
    });
  });
});
```

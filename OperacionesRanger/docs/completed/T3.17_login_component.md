```typescript
import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormBuilder, FormGroup, ReactiveFormsModule, Validators } from '@angular/forms';
import { Router } from '@angular/router';
import { MatCardModule } from '@angular/material/card';
import { MatFormFieldModule } from '@angular/material/form-field';
import { MatInputModule } from '@angular/material/input';
import { MatButtonModule } from '@angular/material/button';
import { MatIconModule } from '@angular/material/icon';
import { MatProgressSpinnerModule } from '@angular/material/progress-spinner';
import { MatSnackBar, MatSnackBarModule } from '@angular/material/snack-bar';

import { AuthService } from '../../../core/services/auth.service';

@Component({
  selector: 'app-login',
  standalone: true,
  imports: [
    CommonModule,
    ReactiveFormsModule,
    MatCardModule,
    MatFormFieldModule,
    MatInputModule,
    MatButtonModule,
    MatIconModule,
    MatProgressSpinnerModule,
    MatSnackBarModule
  ],
  template: `...`,  // Ver archivo completo
  styles: [`...`]   // Ver archivo completo
})
export class LoginComponent implements OnInit {
  loginForm: FormGroup;
  hidePassword = true;
  isLoading = false;

  constructor(
    private fb: FormBuilder,
    private authService: AuthService,
    private router: Router,
    private snackBar: MatSnackBar
  ) {
    this.loginForm = this.fb.group({
      username: ['', [Validators.required]],
      password: ['', [Validators.required]]
    });
  }

  ngOnInit(): void {
    // Si el usuario ya está autenticado, redirigir al dashboard
    if (this.authService.isAuthenticated()) {
      this.router.navigate(['/dashboard']);
    }
  }

  onSubmit(): void {
    if (this.loginForm.invalid) {
      return;
    }

    this.isLoading = true;
    const { username, password } = this.loginForm.value;

    this.authService.login({ username, password }).subscribe({
      next: (response) => {
        this.isLoading = false;
        this.showMessage('Inicio de sesión exitoso', 'success');
        this.router.navigate(['/dashboard']);
      },
      error: (error) => {
        this.isLoading = false;
        const errorMessage = error?.error?.message || 'Error al iniciar sesión. Verifique sus credenciales.';
        this.showMessage(errorMessage, 'error');
        this.loginForm.patchValue({ password: '' });
      }
    });
  }

  private showMessage(message: string, type: 'success' | 'error'): void {
    this.snackBar.open(message, 'Cerrar', {
      duration: type === 'success' ? 3000 : 5000,
      horizontalPosition: 'center',
      verticalPosition: 'top',
      panelClass: type === 'success' ? ['success-snackbar'] : ['error-snackbar']
    });
  }
}
```

### 5. Integración con AuthService

**Flujo de autenticación**:
1. Usuario ingresa credenciales
2. Componente valida formulario
3. Llama a `authService.login(credentials)`
4. AuthService hace POST a `/api/auth/login`
5. Backend responde con tokens y datos del usuario
6. AuthService guarda tokens en localStorage
7. AuthService actualiza estado observable
8. Componente redirige a `/dashboard`

**Datos guardados**:
- `access_token`: JWT de 15 minutos
- `refresh_token`: JWT de 7 días
- `current_user`: Objeto User con id, username, rol, nombre_completo, etc.

**Lógica implementada**:

- **Al cargar componente** (ngOnInit):
  - Si `authService.isAuthenticated()` es true → Redirigir a `/dashboard`
  - Si no autenticado → Mostrar formulario de login

- **Después de login exitoso**:
  - Redirigir a `/dashboard`
  - Nota: En el futuro se puede implementar `returnUrl` del authGuard

### Protección de Rutas

Todas las rutas dentro de LayoutComponent están protegidas por `authGuard`:

1. Usuario intenta acceder a `/dashboard`
2. AuthGuard verifica `authService.isAuthenticated()`
3. Si NO autenticado → Redirige a `/login`
4. Si autenticado → Permite acceso
5. Usuario navega libremente por el sistema

### Expiración de Tokens

**Access Token (15 minutos)**:
- AuthInterceptor verifica expiración antes de cada request
- Si está por expirar (<5 min), llama a `authService.refreshToken()`
- Refresh exitoso → Continúa request con nuevo token
- Refresh falla → Redirige a login

**Refresh Token (7 días)**:
- Si refresh token expira, usuario debe volver a hacer login
- Backend invalida refresh token al hacer logout

### JWT Token Structure

**Header**:
```json
{
  "alg": "HS256",
  "typ": "JWT"
}
```

**Payload**:
```json
{
  "sub": 1,
  "username": "admin",
  "rol": "ADMIN",
  "iat": 1642592400,
  "exp": 1642593300
}
```

**Signature**:
```
HMACSHA256(
  base64UrlEncode(header) + "." + base64UrlEncode(payload),
  JWT_SECRET
)
```
